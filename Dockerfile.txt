# Stage 1: Build phase to compile our high-performance architecture
FROM node:20-alpine AS builder
WORKDIR /app
COPY package*.json ./
RUN npm install
COPY . .
RUN npx tsc

# Stage 2: Minimalist production runtime phase
FROM node:20-alpine AS runner
WORKDIR /app
ENV NODE_ENV=production

COPY package*.json ./
RUN npm install --only=production

# Pull the compiled server distributions from the builder slice safely
COPY --from=builder /app/dist ./dist
COPY --from=builder /app/src/crypto ./dist/crypto

EXPOSE 8080
CMD ["node", "dist/server/index.js"]
